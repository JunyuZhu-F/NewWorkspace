cmake_minimum_required(VERSION 3.20)
project(scale_directory_admission LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
set(SCALE_PYTHON "D:/anaconda3/python.exe" CACHE FILEPATH "Existing Python for source generation")
execute_process(COMMAND "${SCALE_PYTHON}" -B "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${CMAKE_CURRENT_SOURCE_DIR}/../wiki_experiment/directory.cpp" "${CMAKE_BINARY_DIR}/scale_directory.cpp" RESULT_VARIABLE GEN_STATUS)
if(NOT GEN_STATUS EQUAL 0)
  message(FATAL_ERROR "Scale admission generation failed")
endif()
add_executable(scale_directory "${CMAKE_BINARY_DIR}/scale_directory.cpp" ../tests/oracle.cpp)
target_include_directories(scale_directory PRIVATE ../tests)
target_link_libraries(scale_directory PRIVATE directory_core)
if(MSVC)
  target_compile_options(scale_directory PRIVATE /W4 /permissive- /utf-8)
  if(DIRECTORY_ENABLE_ASAN)
    target_compile_options(scale_directory PRIVATE /fsanitize=address)
    target_link_options(scale_directory PRIVATE /INCREMENTAL:NO)
  endif()
endif()
