cmake_minimum_required(VERSION 3.20)
project(scale_formal LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
execute_process(COMMAND "D:/anaconda3/python.exe" -B "${CMAKE_CURRENT_SOURCE_DIR}/generate.py" "${CMAKE_CURRENT_SOURCE_DIR}/../wiki_experiment/directory.cpp" "${CMAKE_BINARY_DIR}/scale_formal.cpp" RESULT_VARIABLE GEN_STATUS)
if(NOT GEN_STATUS EQUAL 0)
  message(FATAL_ERROR "Formal generation failed")
endif()
add_executable(scale_formal "${CMAKE_BINARY_DIR}/scale_formal.cpp" ../tests/oracle.cpp)
target_include_directories(scale_formal PRIVATE ../tests)
target_link_libraries(scale_formal PRIVATE directory_core)
if(MSVC)
  target_compile_options(scale_formal PRIVATE /W4 /permissive- /utf-8)
  if(DIRECTORY_ENABLE_ASAN)
    target_compile_options(scale_formal PRIVATE /fsanitize=address)
    target_link_options(scale_formal PRIVATE /INCREMENTAL:NO)
  endif()
endif()
