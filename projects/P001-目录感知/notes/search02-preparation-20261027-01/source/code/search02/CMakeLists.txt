cmake_minimum_required(VERSION 3.20)
project(directory_search02 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
# Reuse the verified backend and contracts verbatim without editing frozen sources.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../search/search.cpp" SEARCH01_SOURCE)
string(FIND "${SEARCH01_SOURCE}" "int main(" MAIN_POSITION)
if(MAIN_POSITION LESS 1)
  message(FATAL_ERROR "SEARCH-01 source entry marker missing")
endif()
string(SUBSTRING "${SEARCH01_SOURCE}" 0 ${MAIN_POSITION} SEARCH01_LIBRARY)
file(WRITE "${CMAKE_BINARY_DIR}/search01_backend.inc" "${SEARCH01_LIBRARY}")
add_executable(directory_search02 benchmark.cpp)
target_include_directories(directory_search02 PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(directory_search02 PRIVATE directory_core)
if(MSVC)
  target_compile_options(directory_search02 PRIVATE /W4 /permissive- /utf-8 /fp:strict)
  if(DIRECTORY_ENABLE_ASAN)
    target_compile_options(directory_search02 PRIVATE /fsanitize=address)
    target_link_options(directory_search02 PRIVATE /INCREMENTAL:NO)
  endif()
endif()
enable_testing()
add_test(NAME search02_contract COMMAND directory_search02 self-test)
set_tests_properties(search02_contract PROPERTIES TIMEOUT 60)
