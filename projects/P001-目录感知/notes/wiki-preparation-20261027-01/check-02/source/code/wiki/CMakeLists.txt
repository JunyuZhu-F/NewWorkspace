cmake_minimum_required(VERSION 3.20)
project(directory_wiki_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
add_executable(directory_wiki_validate validate.cpp ../tests/oracle.cpp)
target_include_directories(directory_wiki_validate PRIVATE ../tests)
target_link_libraries(directory_wiki_validate PRIVATE directory_core)
if(MSVC)
  target_compile_options(directory_wiki_validate PRIVATE /W4 /permissive- /utf-8)
  if(DIRECTORY_ENABLE_ASAN)
    target_compile_options(directory_wiki_validate PRIVATE /fsanitize=address)
    target_link_options(directory_wiki_validate PRIVATE /INCREMENTAL:NO)
  endif()
endif()
