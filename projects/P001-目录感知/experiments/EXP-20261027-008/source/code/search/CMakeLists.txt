cmake_minimum_required(VERSION 3.20)
project(directory_search_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
add_executable(directory_search search.cpp)
target_link_libraries(directory_search PRIVATE directory_core)
if(MSVC)
  target_compile_options(directory_search PRIVATE /W4 /permissive- /utf-8 /fp:strict)
  if(DIRECTORY_ENABLE_ASAN)
    target_compile_options(directory_search PRIVATE /fsanitize=address)
    target_link_options(directory_search PRIVATE /INCREMENTAL:NO)
  endif()
endif()
enable_testing()
add_test(NAME search_contract COMMAND directory_search self-test)
set_tests_properties(search_contract PROPERTIES TIMEOUT 60)
