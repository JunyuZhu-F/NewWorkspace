cmake_minimum_required(VERSION 3.20)
project(directory_reproduction VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(croaring STATIC third_party/croaring/roaring.c)
target_include_directories(croaring SYSTEM PUBLIC third_party/croaring)
add_library(directory_core src/path.cpp src/index.cpp)
target_include_directories(directory_core PUBLIC include)
target_link_libraries(directory_core PUBLIC croaring)
# Same source, separate diagnostic target. The default core has no counter member/path.
add_library(directory_core_instrumented src/path.cpp src/index.cpp)
target_include_directories(directory_core_instrumented PUBLIC include)
target_link_libraries(directory_core_instrumented PUBLIC croaring)
target_compile_definitions(directory_core_instrumented PUBLIC DIRECTORY_ENABLE_WORK_COUNTERS)
if(MSVC)
  target_compile_options(directory_core PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(croaring PRIVATE /utf-8)
  target_compile_options(directory_core_instrumented PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(directory_core_tests tests/oracle.cpp tests/core_tests.cpp)
target_link_libraries(directory_core_tests PRIVATE directory_core)
if(MSVC)
  target_compile_options(directory_core_tests PRIVATE /W4 /permissive- /utf-8)
endif()
add_executable(directory_benchmark tests/oracle.cpp tests/benchmark.cpp)
target_link_libraries(directory_benchmark PRIVATE directory_core)
add_executable(directory_benchmark_instrumented tests/oracle.cpp tests/benchmark.cpp)
target_link_libraries(directory_benchmark_instrumented PRIVATE directory_core_instrumented)
if(MSVC)
  target_compile_options(directory_benchmark PRIVATE /W4 /permissive- /utf-8)
  target_compile_options(directory_benchmark_instrumented PRIVATE /W4 /permissive- /utf-8)
endif()
if(WIN32)
  target_link_libraries(directory_benchmark PRIVATE Psapi)
  target_link_libraries(directory_benchmark_instrumented PRIVATE Psapi)
endif()
enable_testing()
add_test(NAME bitmap_contract COMMAND directory_core_tests bitmap)
add_test(NAME fixed_semantics COMMAND directory_core_tests fixed)
foreach(seed IN ITEMS 7 42 2026 65536)
  add_test(NAME random_${seed} COMMAND directory_core_tests random ${seed} 1000)
endforeach()
set_tests_properties(bitmap_contract fixed_semantics random_7 random_42 random_2026 random_65536 PROPERTIES TIMEOUT 120)

option(DIRECTORY_ENABLE_ASAN "Instrument core correctness tests with AddressSanitizer" OFF)
if(DIRECTORY_ENABLE_ASAN)
  if(MSVC)
    foreach(target IN ITEMS croaring directory_core directory_core_instrumented directory_core_tests directory_benchmark directory_benchmark_instrumented)
      target_compile_options(${target} PRIVATE /fsanitize=address)
    endforeach()
    target_link_options(directory_core_tests PRIVATE /INCREMENTAL:NO)
    target_link_options(directory_benchmark PRIVATE /INCREMENTAL:NO)
    target_link_options(directory_benchmark_instrumented PRIVATE /INCREMENTAL:NO)
  else()
    foreach(target IN ITEMS croaring directory_core directory_core_instrumented directory_core_tests directory_benchmark directory_benchmark_instrumented)
      target_compile_options(${target} PRIVATE -fsanitize=address -fno-omit-frame-pointer)
    endforeach()
    target_link_options(directory_core_tests PRIVATE -fsanitize=address)
    target_link_options(directory_benchmark PRIVATE -fsanitize=address)
    target_link_options(directory_benchmark_instrumented PRIVATE -fsanitize=address)
  endif()
endif()
set(DIRECTORY_PILOT_TSV "" CACHE FILEPATH "Optional frozen time-pilot correctness fixture")
if(DIRECTORY_PILOT_TSV)
  add_test(NAME time_pilot_correctness COMMAND directory_core_tests pilot "${DIRECTORY_PILOT_TSV}")
  set_tests_properties(time_pilot_correctness PROPERTIES TIMEOUT 120)
endif()
