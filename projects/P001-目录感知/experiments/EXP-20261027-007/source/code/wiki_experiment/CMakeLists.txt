cmake_minimum_required(VERSION 3.20)
project(wiki_independent_experiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." "${CMAKE_BINARY_DIR}/directory")
add_library(wiki_phase_core ../src/path.cpp ../src/index.cpp ../src/diagnostics.cpp)
target_include_directories(wiki_phase_core PUBLIC ../include)
target_link_libraries(wiki_phase_core PUBLIC croaring)
target_compile_definitions(wiki_phase_core PUBLIC DIRECTORY_PHASE_DIAGNOSTICS)
add_executable(wiki_directory directory.cpp ../tests/oracle.cpp)
add_executable(wiki_directory_phase directory.cpp ../tests/oracle.cpp)
target_include_directories(wiki_directory PRIVATE ../tests)
target_include_directories(wiki_directory_phase PRIVATE ../tests)
target_link_libraries(wiki_directory PRIVATE directory_core)
target_link_libraries(wiki_directory_phase PRIVATE wiki_phase_core)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../search/search.cpp" BACKEND)
string(FIND "${BACKEND}" "int main(" MAIN_POSITION)
string(SUBSTRING "${BACKEND}" 0 ${MAIN_POSITION} BACKEND_LIBRARY)
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/search01_backend.inc" CONTENT "${BACKEND_LIBRARY}" NEWLINE_STYLE UNIX)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../search02/benchmark.cpp" SEARCH)
string(REPLACE "queries.n==501" "queries.n==456" SEARCH "${SEARCH}")
string(REPLACE "cases.size()==1002" "cases.size()==912" SEARCH "${SEARCH}")
string(REPLACE "search02_contract" "wiki_search_contract" SEARCH "${SEARCH}")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/wiki_search.cpp" CONTENT "${SEARCH}" NEWLINE_STYLE UNIX)
add_executable(wiki_search "${CMAKE_BINARY_DIR}/wiki_search.cpp")
target_include_directories(wiki_search PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(wiki_search PRIVATE directory_core)
if(MSVC)
  foreach(t IN ITEMS wiki_directory wiki_directory_phase wiki_phase_core wiki_search)
    target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8 /fp:strict)
    if(DIRECTORY_ENABLE_ASAN)
      target_compile_options(${t} PRIVATE /fsanitize=address)
    endif()
  endforeach()
  if(DIRECTORY_ENABLE_ASAN)
    foreach(t IN ITEMS wiki_directory wiki_directory_phase wiki_search)
      target_link_options(${t} PRIVATE /INCREMENTAL:NO)
    endforeach()
  endif()
endif()
